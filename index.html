Below is my JavaScript solution for the Codewars problem "Merge two sorted arrays into one":

function mergeArrays(arr1, arr2) {
  if (arr1.length === 0 && arr2.length === 0) return [];
  
  let sortedArr = arr1.concat(arr2).sort((a, b) => a - b);
  let uniqueSortedArr = [...new Set(sortedArr)];
  
  return uniqueSortedArr;
    }
